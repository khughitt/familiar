# Phase 4 `theme add` Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** `familiar theme add <https-url|path>` — bounded acquisition,
validation, atomic promotion, external receipts — plus a receipt-aware
`familiar theme list`.

**Architecture:** Three new engine modules under `src/theme/` (receipt store,
acquisition, orchestrator), one catalog extension, CLI wiring in
`bin/familiar`. The orchestrator serializes installers with the existing
`withLock` and implements the conformance-gate §7 sequence exactly:
acquire into `userThemesDir/.staging/<run>` → provenance → strip `.git` →
validate the staged bytes → prove target and receipt absent → atomic
rename → receipt last.

**Tech Stack:** Node ≥ 20 ESM, `node:test`, stdlib only (no new
dependencies). `familiar-theme` v0.1.0 provides `validateThemePack` and
`LIMITS`. Hermetic HTTPS tests use a committed self-signed localhost cert.

**Spec:** `docs/specs/2026-08-18-theme-add-design.md` — read it first; this
plan argues from it. It cites the archive's conformance-gate spec §7 as the
normative sequence.

## Global Constraints

- Conventional commits; **no AI-attribution trailers** anywhere.
- Explicit > defensive; fail early; no silent fallbacks. Every failure is a
  named, single-line instruction.
- No "legacy"/"compatibility" layers; no `Unified` prefixes.
- The contract package (`familiar-theme`) is **not** modified — installation
  is engine policy.
- Absolute machine paths (`/home/...`, `/mnt/...`) never appear in code
  comments or docs.
- Fast suite green after every task: `npm test`. All new tests are fast
  (hermetic, no network).
- Work in the `redesign-4` worktree; base is `main` at `be32dd0` or later.

```js
import { readFileSync } from 'node:fs';
import { join, isAbsolute } from 'node:path';
import { writeJsonAtomic } from '../bus/store.js';

// The receipt is the external claim "these bytes passed validation, acquired
// from here, then". It lives OUTSIDE userThemesDir because the catalog
// treats every directory there as a theme id, and outside the pack because
// the pack's own bytes must be exactly the validated bytes (spec §2).
//
// Reading is strict and never throws on content: a receipt that cannot make
// its claim precisely is reported as invalid WITH the reason, because an
// invalid receipt collapsed into 'absent' would make corruption look like a
// clean manual install, and one collapsed into 'validated' would be a false
// claim.
export function receiptPath(paths, id) {
  return join(paths.themeReceiptsDir, `${id}.json`);
}

export async function writeReceipt(paths, receipt) {
  await writeJsonAtomic(receiptPath(paths, receipt.id), receipt);
}

const HEX40 = /^[0-9a-f]{40}$/;

function invalidReason(data, id) {
  if (typeof data !== 'object' || data === null) return 'not an object';
  if (data.id !== id) return `embedded id ${JSON.stringify(data.id)} does not match the filename`;
  const { source } = data;
  if (typeof source !== 'object' || source === null) return 'source is not an object';
  if (source.kind === 'https') {
    if (typeof source.url !== 'string' || !source.url.startsWith('https://')) return 'source.url is not an https URL';
    if (typeof source.commit !== 'string' || !HEX40.test(source.commit)) return 'source.commit is not a 40-hex sha';
  } else if (source.kind === 'local') {
    if (typeof source.path !== 'string' || !isAbsolute(source.path)) return 'source.path is not an absolute path';
  } else {
    return `unknown source.kind ${JSON.stringify(source.kind)}`;
  }
  if (typeof data.installedAt !== 'string' || Number.isNaN(Date.parse(data.installedAt))) {
    return 'installedAt is not an ISO-8601 date';
  }
  return null;
}

export function readReceipt(paths, id, { readFile = readFileSync } = {}) {
  let text;
  try {
    text = readFile(receiptPath(paths, id), 'utf8');
  } catch (error) {
    if (error.code === 'ENOENT') return { verdict: 'absent' };
    throw error;
  }
  let data;
  try {
    data = JSON.parse(text);
  } catch (error) {
    return { verdict: 'invalid', reason: `unparseable JSON: ${error.message}` };
  }
  const reason = invalidReason(data, id);
  if (reason !== null) return { verdict: 'invalid', reason };
  return { verdict: 'validated', receipt: data };
}
```

Note `writeJsonAtomic` must create the parent directory — check
`src/bus/store.js`: if it does not `mkdir` the parent, do it in
`writeReceipt` (`mkdirSync(paths.themeReceiptsDir, { recursive: true })`

```js
import { statSync } from 'node:fs';
import { resolve } from 'node:path';
import { stripVTControlCharacters } from 'node:util';

// Transport rule (spec §1): HTTPS URLs and local directories, nothing else.
// Credentialed URLs are refused because the given URL is persisted in the
// receipt and printed by `theme list` — accepting one stores and displays a
// secret. Private repos use a local clone plus `theme add ./dir`.
export function classifySource(raw, { stat = statSync } = {}) {
  if (raw.startsWith('https://')) {
    const url = new URL(raw);
    if (url.username !== '' || url.password !== '') {
      throw new Error(
        'theme add: credentials in the URL would be stored in the receipt and shown by `theme list` — use a credential-free HTTPS URL, or clone privately and add the local directory'
      );
    }
    return { kind: 'https', url: raw };
  }
  if (/^[a-z][a-z0-9+.-]*:\/\//i.test(raw) || /^git@/.test(raw)) {
    throw new Error(
      `theme add: sources are HTTPS URLs or local directories — got ${JSON.stringify(raw)}`
    );
  }
  const path = resolve(raw);
  let st;
  try {
    st = stat(path);
  } catch (error) {
    if (error.code === 'ENOENT') throw new Error(`theme add: no directory at ${path}`);
    throw error;
  }
  if (!st.isDirectory()) throw new Error(`theme add: ${path} is not a directory`);
  return { kind: 'local', path };
}

// Remote git diagnostics are untrusted terminal input. ANSI stripping alone
// leaves bare \r (git progress uses it), backspace and BEL live, any of
// which can rewrite the visible line — so after stripping ANSI, CR/LF both
// split lines and every remaining C0/C1 control is replaced (spec §4).
export function collapseStderr(text) {
  return stripVTControlCharacters(String(text))
    .split(/\r\n|\r|\n/)
    .map((line) => line.replaceAll(/\p{Cc}/gu, ' ').trim())
    .filter((line) => line.length > 0)
    .join('; ');
}
```

```js
// The defensive copy (spec §3). Validation runs only AFTER acquisition, so
// the copy itself must refuse what it cannot safely materialize: it walks
// with lstat and never dereferences — a symlink, FIFO, socket or device is
// rejected by path, never opened, never recreated in staging. `.git` at any
// depth is excluded; git is never invoked on the source. Regular files are
// copied by abortable streaming because fs.copyFile/fs.cp accept no
// AbortSignal, and the wall clock must be able to stop a stalled file.
export async function copySource(sourceDir, dest, { signal } = {}) {
  const sourceReal = realpathSync(sourceDir);
  const destReal = realpathSync(dest);
  if (destReal === sourceReal || destReal.startsWith(sourceReal + sep)) {
    throw new Error(
      `theme add: staging at ${destReal} lies inside the source ${sourceReal} — a self-copy would never terminate`
    );
  }
  await copyTree(sourceReal, destReal, signal);
}

async function copyTree(from, to, signal) {
  const entries = readdirSync(from, { withFileTypes: true })
    .sort((a, b) => (a.name < b.name ? -1 : 1));
  for (const entry of entries) {
    if (entry.name === '.git') continue;
    const src = join(from, entry.name);
    const out = join(to, entry.name);
    const st = lstatSync(src);
    if (st.isDirectory()) {
      mkdirSync(out);
      await copyTree(src, out, signal);
    } else if (st.isFile()) {
      try {
        await pipeline(createReadStream(src), createWriteStream(out, { flags: 'wx' }), { signal });
      } catch (error) {
        throw signal?.aborted ? signal.reason : error;
      }
    } else {
      throw new Error(
        `theme add: ${src} is not a regular file or directory — the source must contain only files and directories`
      );
    }
  }
}
```

`signal?.aborted` mapping: an already-aborted signal makes `pipeline`
reject with the signal's reason wrapped — unwrapping to `signal.reason`
keeps the bound's named error the visible one.

```js
export const DEFAULT_TIMEOUT_MS = 300_000;
export const DEFAULT_GROWTH_LIMIT_BYTES = 4 * LIMITS.MAX_TOTAL_BYTES;

// Git isolation (spec §3). The machine's own configuration must not shape
// what an untrusted clone does — this host's global filter.lfs.required is
// the standing proof. Isolation is applied AFTER caller input, so the one
// test seam (caFile → GIT_SSL_CAINFO) cannot weaken it.
export function buildCloneEnv({ caFile } = {}, base = process.env) {
  const env = { ...base };
  if (caFile !== undefined) env.GIT_SSL_CAINFO = caFile;
  env.GIT_CONFIG_GLOBAL = devNull;
  env.GIT_CONFIG_SYSTEM = devNull;
  env.GIT_TERMINAL_PROMPT = '0';
  env.GIT_ASKPASS = '';
  env.SSH_ASKPASS = '';
  env.GIT_ALLOW_PROTOCOL = 'https';
  return env;
}

function runGit(args, { env, signal, what }) {
  return new Promise((resolveRun, rejectRun) => {
    const child = spawn('git', args, { env, stdio: ['ignore', 'pipe', 'pipe'] });
    let stdout = '';
    let stderr = '';
    child.stdout.on('data', (chunk) => { stdout += chunk; });
    child.stderr.on('data', (chunk) => { stderr += chunk; });
    const onAbort = () => child.kill('SIGKILL');
    signal?.addEventListener('abort', onAbort, { once: true });
    child.on('error', rejectRun);
    child.on('close', (code) => {
      signal?.removeEventListener('abort', onAbort);
      if (signal?.aborted) return rejectRun(signal.reason);
      if (code !== 0) return rejectRun(new Error(`${what} failed: ${collapseStderr(stderr)}`));
      resolveRun(stdout);
    });
    if (signal?.aborted) onAbort();
  });
}

export async function cloneSource(url, dest, { caFile, signal } = {}) {
  const env = buildCloneEnv({ caFile });
  await runGit(
    ['clone', '--depth', '1', '--single-branch', '--no-tags', '--no-recurse-submodules', '--', url, dest],
    { env, signal, what: 'clone' });
  // Provenance is read while the clone's .git still exists (spec §2 step 3);
  // the orchestrator removes .git right after this returns.
  const commit = (await runGit(['-C', dest, 'rev-parse', 'HEAD'], { env, signal, what: 'rev-parse' })).trim();
  return { commit };
}

// The honest fetch bound (spec §3): git has no client-side aggregate
// fetch-byte limit, so v1 is a wall clock plus a growth monitor with
// bounded overshoot; the exact limits apply at validation. The monitor's
// traversal is lstat-based and never follows symlinks — a clone's contents
// are untrusted before validation, and a symlink must not redirect the scan
// outside staging.
export async function acquireSource(source, dest, {
  caFile,
  timeoutMs = DEFAULT_TIMEOUT_MS,
  growthLimitBytes = DEFAULT_GROWTH_LIMIT_BYTES,
  pollMs = 1000,
} = {}) {
  const controller = new AbortController();
  const timer = setTimeout(() => {
    controller.abort(new Error(`theme add: acquisition exceeded the ${timeoutMs} ms wall clock`));
  }, timeoutMs);
  const poller = setInterval(() => {
    const bytes = stagedBytes(dest);
    if (bytes > growthLimitBytes) {
      controller.abort(new Error(
        `theme add: staging grew past the ${growthLimitBytes}-byte bound (${bytes} bytes fetched)`));
    }
  }, pollMs);
  try {
    if (source.kind === 'https') {
      const { commit } = await cloneSource(source.url, dest, { caFile, signal: controller.signal });
      return { kind: 'https', url: source.url, commit };
    }
    const path = realpathSync(source.path);
    await copySource(path, dest, { signal: controller.signal });
    return { kind: 'local', path };
  } catch (error) {
    throw controller.signal.aborted ? controller.signal.reason : error;
  } finally {
    clearTimeout(timer);
    clearInterval(poller);
  }
}

function stagedBytes(dir) {
  let total = 0;
  let entries;
  // Entries appear and vanish mid-clone; a name that disappeared between
  // readdir and lstat is simply not counted this poll.
  try {
    entries = readdirSync(dir, { withFileTypes: true });
  } catch {
    return total;
  }
  for (const entry of entries) {
    const path = join(dir, entry.name);
    let st;
    try {
      st = lstatSync(path);
    } catch {
      continue;
    }
    total += st.size;
    if (st.isDirectory()) total += stagedBytes(path);
  }
  return total;
}
```

```js
import {
  lstatSync, mkdirSync, mkdtempSync, readdirSync, renameSync, rmSync, existsSync,
} from 'node:fs';
import { join } from 'node:path';
import { validateThemePack } from 'familiar-theme';
import { withLock } from '../bus/lock.js';
import { classifySource, acquireSource } from './acquire.js';
import { receiptPath, writeReceipt } from './receipt.js';

export const STAGING_DIR_NAME = '.staging';

// The lock lives WITH the resource it protects: configDir and stateDir are
// independently overridable, so a lock anywhere else would let two
// processes share the themes directory while holding different locks. It is
// a lock FILE, which the catalog ignores (only directories are theme
// candidates).
export function lockPathFor(paths) {
  return join(paths.userThemesDir, '.theme-add.lock');
}

// The §7 sequence, serialized. staleMs is Infinity because no finite value
// is safe: validation has no enforced time ceiling, so any number is an
// estimate that reclaims a slow LIVE installer; dead and pid-recycled
// holders are already reclaimed immediately by the liveness check, and a
// live wedged holder is the user's to kill, not a reclaimer's to guess
// about.
export async function addTheme({
  paths, source, caFile, timeoutMs, growthLimitBytes,
  lockOpts = {},
  validate = validateThemePack,
  writeReceiptFn = writeReceipt,
  now = () => new Date(),
}) {
  const classified = classifySource(source);
  mkdirSync(paths.userThemesDir, { recursive: true });
  try {
    return await withLock(lockPathFor(paths), async () => {
      const stagingRoot = join(paths.userThemesDir, STAGING_DIR_NAME);
      let st = null;
      try {
        st = lstatSync(stagingRoot);
      } catch (error) {
        if (error.code !== 'ENOENT') throw error;
      }
      if (st === null) mkdirSync(stagingRoot);
      else if (!st.isDirectory()) {
        throw new Error(`theme add: ${stagingRoot} is not a directory — remove it`);
      }
      // Under the lock, any entry here is abandoned: no live run can exist
      // while we hold it (spec §2 step 1).
      for (const entry of readdirSync(stagingRoot)) {
        rmSync(join(stagingRoot, entry), { recursive: true, force: true });
      }
      const runDir = mkdtempSync(join(stagingRoot, 'run-'));
      let promoted = false;
      try {
        const provenance = await acquireSource(classified, runDir,
          { caFile, timeoutMs, growthLimitBytes });
        rmSync(join(runDir, '.git'), { recursive: true, force: true });
        const pack = await validate(runDir);
        const target = join(paths.userThemesDir, pack.id);
        // lstat, not exists: an empty directory or a dangling symlink at the
        // target must refuse too — rename would silently replace an empty
        // directory (spec §2 step 6).
        let targetSt = null;
        try {
          targetSt = lstatSync(target);
        } catch (error) {
          if (error.code !== 'ENOENT') throw error;
        }
        if (targetSt !== null) {
          throw new Error(`theme '${pack.id}' is already installed at ${target} — remove it first`);
        }
        // An orphan receipt would survive a crash after the rename and then
        // describe the NEW bytes — a false validation claim. Proving it
        // absent here keeps receipt-last unconditional (spec §2 step 6).
        const receiptFile = receiptPath(paths, pack.id);
        if (existsSync(receiptFile)) {
          throw new Error(
            `theme add: orphan receipt at ${receiptFile} describes no installed theme — remove it first`);
        }
        renameSync(runDir, target);
        promoted = true;
        const receipt = { id: pack.id, source: provenance, installedAt: now().toISOString() };
        await writeReceiptFn(paths, receipt);
        return { id: pack.id, dir: target, members: pack.members.size, provenance };
      } finally {
        // One owner for staging cleanup: every error path — acquisition,
        // validation, refusals, a failed rename — removes the run; only a
        // crash leaves it, for the next add (spec §2).
        if (!promoted) rmSync(runDir, { recursive: true, force: true });
      }
    }, { staleMs: Infinity, ...lockOpts });
  } catch (error) {
    if (/^could not acquire lock:/.test(error.message)) {
      throw new Error('theme add: another theme add is running');
    }
    throw error;
  }
}
```

```js
// The list-side view of installer staging. This is a read of untrusted
// filesystem state, so it is guarded like the installer's own: lstat first,
// and only a real directory is ever read — a symlink or file at .staging
// must neither redirect the read outside the themes root nor crash listing
// with ENOTDIR. 'empty' is the steady state after a successful install
// (each add creates the parent; cleanup removes only run entries) and
// reports nothing.
export function stagingStatus(paths, { lstat = lstatSync, readdir = readdirSync } = {}) {
  const dir = join(paths.userThemesDir, STAGING_DIR_NAME);
  let st;
  try {
    st = lstat(dir);

    if (error.code === 'ENOENT') return 'absent';
    throw error;
  }
  if (!st.isDirectory()) return 'not-a-directory';
  return readdir(dir).length > 0 ? 'occupied' : 'empty';
}
```

```js
const GIT_ENV = {
  ...process.env,
  GIT_CONFIG_GLOBAL: devNull, GIT_CONFIG_SYSTEM: devNull,

// A bare, dumb-HTTP-servable repo whose work tree is `dir`.
function bareRepoFrom(dir) {
  git(['init', '-q', '-b', 'main'], dir);
  git(['add', '-A'], dir);
  git(['commit', '-q', '-m', 'fixture pack'], dir);
  const bare = join(mkdtempSync(join(tmpdir(), 'bare-')), 'repo.git');
  execFileSync('git', ['clone', '-q', '--bare', dir, bare], { env: GIT_ENV });
  git(['update-server-info'], bare);
  return bare;
}

// Static file server over the bare repo: git's dumb HTTP protocol needs
// nothing more. Real TLS, real https URL, production protocol allowlist.
async function serveBare(bareDir) {
  const server = createServer(
    { cert: readFileSync(CERT), key: readFileSync(KEY) },
    (req, res) => {
      const rel = normalize(decodeURIComponent(new URL(req.url, 'https://x').pathname)).replace(/^[/\\]+/, '');
      const file = join(bareDir, rel);
      // 404 anything that is not a readable file under the repo — including
      // directories, whose readFileSync would otherwise throw EISDIR and
      // crash the server mid-test.
      try {
        if (!file.startsWith(bareDir + sep)) throw new Error('outside root');
        res.end(readFileSync(file));
      } catch {
        res.statusCode = 404;
        res.end();
      }
    });
  await new Promise((ready) => server.listen(0, '127.0.0.1', ready));
  return {
    url: `https://127.0.0.1:${server.address().port}`,
    close: () => new Promise((closed) => server.close(closed)),
  };
}
```
